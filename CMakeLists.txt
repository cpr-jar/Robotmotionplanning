cmake_minimum_required(VERSION 3.20)
project(robot_motion VERSION 0.13.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(MOTION_IS_TOP_LEVEL OFF)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  set(MOTION_IS_TOP_LEVEL ON)
endif()
set(MOTION_TESTS_DEFAULT ${MOTION_IS_TOP_LEVEL})
if(DEFINED BUILD_TESTING AND NOT BUILD_TESTING)
  set(MOTION_TESTS_DEFAULT OFF)
endif()
option(MOTION_BUILD_TESTS "Build motion tests" ${MOTION_TESTS_DEFAULT})
option(MOTION_BUILD_EXAMPLES "Build standalone motion examples" ${MOTION_IS_TOP_LEVEL})
option(MOTION_WITH_SERIAL "Build serial solver, model/program import and simulator" ON)
option(MOTION_WITH_KDL_REFERENCE "Build independent, test-only KDL FK/J comparison" ON)
option(MOTION_WITH_RUCKIG "Build optional local Ruckig adapter" OFF)
option(MOTION_SANITIZE "Enable address/undefined sanitizers" OFF)
option(MOTION_BUILD_FUZZER "Build Clang libFuzzer program parser (requires sanitizers)" OFF)

if(MOTION_BUILD_FUZZER)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR NOT MOTION_SANITIZE OR NOT MOTION_WITH_SERIAL OR NOT MOTION_BUILD_TESTS)
    message(FATAL_ERROR "MOTION_BUILD_FUZZER requires Clang, MOTION_SANITIZE=ON, MOTION_WITH_SERIAL=ON and MOTION_BUILD_TESTS=ON")
  endif()
  add_compile_options(-fsanitize=fuzzer-no-link)
  add_link_options(-fsanitize=fuzzer-no-link)
endif()
if(MOTION_SANITIZE AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

include(cmake/MotionTargets.cmake)
include(cmake/MotionDependencies.cmake)
add_library(robot_motion INTERFACE)
add_library(RobotMotion::motion ALIAS robot_motion)
target_include_directories(robot_motion INTERFACE ${PROJECT_SOURCE_DIR}/include)
target_compile_features(robot_motion INTERFACE cxx_std_20)
if(MOTION_SANITIZE AND NOT MSVC)
  # Consumers outside this directory also need the sanitizer compiler/link contract.
  target_compile_options(robot_motion INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(robot_motion INTERFACE -fsanitize=address,undefined)
endif()
if(MOTION_WITH_SERIAL)
  motion_dependencies()
  add_subdirectory(src)
  target_link_libraries(robot_motion INTERFACE robot_motion_serial)
endif()
if(MOTION_WITH_RUCKIG)
  motion_ruckig_dependency()
  target_link_libraries(robot_motion INTERFACE ruckig)
  target_compile_definitions(robot_motion INTERFACE MOTION_WITH_RUCKIG=1)
endif()
if(MOTION_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
if(MOTION_BUILD_EXAMPLES)
  add_subdirectory(examples)
endif()
