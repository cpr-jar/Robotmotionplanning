cmake_minimum_required(VERSION 3.20)
project(robot_motion VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MOTION_WITH_RUCKIG "Enable Ruckig online joint trajectory generator" OFF)
add_library(robot_motion INTERFACE)
target_include_directories(robot_motion INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MOTION_WITH_RUCKIG)
  include(FetchContent)
  FetchContent_Declare(ruckig GIT_REPOSITORY https://github.com/pantor/ruckig.git GIT_TAG v0.15.3 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(ruckig)
  target_link_libraries(robot_motion INTERFACE ruckig)
  target_compile_definitions(robot_motion INTERFACE MOTION_WITH_RUCKIG=1)
  add_executable(online_ptp examples/online_ptp.cpp)
  target_link_libraries(online_ptp PRIVATE robot_motion)
endif()
enable_testing()
add_executable(motion_tests tests/test_core.cpp)
target_link_libraries(motion_tests PRIVATE robot_motion)
add_test(NAME motion_tests COMMAND motion_tests)
add_executable(trajectory_tests tests/test_trajectory.cpp)
target_link_libraries(trajectory_tests PRIVATE robot_motion)
add_test(NAME trajectory_tests COMMAND trajectory_tests)
