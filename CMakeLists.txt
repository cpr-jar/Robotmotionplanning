cmake_minimum_required(VERSION 3.20)
project(robot_motion VERSION 0.12.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MOTION_WITH_SERIAL "Build serial solver, model/program import and simulator" ON)
option(MOTION_WITH_KDL_REFERENCE "Build independent, test-only KDL FK/J comparison" ON)
option(MOTION_WITH_RUCKIG "Build optional local Ruckig adapter" OFF)
option(MOTION_SANITIZE "Enable address/undefined sanitizers" OFF)
option(MOTION_BUILD_FUZZER "Build Clang libFuzzer program parser (requires sanitizers)" OFF)
if(MOTION_BUILD_FUZZER)
 if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR NOT MOTION_SANITIZE OR NOT MOTION_WITH_SERIAL)
  message(FATAL_ERROR "MOTION_BUILD_FUZZER requires Clang, MOTION_SANITIZE=ON and MOTION_WITH_SERIAL=ON")
 endif()
 add_compile_options(-fsanitize=fuzzer-no-link)
 add_link_options(-fsanitize=fuzzer-no-link)
endif()
if(MOTION_SANITIZE AND NOT MSVC)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
add_library(robot_motion INTERFACE)
target_include_directories(robot_motion INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(robot_motion INTERFACE cxx_std_20)

  set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
