cmake_minimum_required(VERSION 3.20)
project(robot_motion VERSION 0.16.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(MOTION_IS_TOP_LEVEL OFF)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  set(MOTION_IS_TOP_LEVEL ON)
endif()
set(MOTION_TESTS_DEFAULT ${MOTION_IS_TOP_LEVEL})
if(DEFINED BUILD_TESTING AND NOT BUILD_TESTING)
  set(MOTION_TESTS_DEFAULT OFF)
endif()
option(MOTION_BUILD_TESTS "Build motion tests" ${MOTION_TESTS_DEFAULT})
option(MOTION_BUILD_EXAMPLES "Build standalone motion examples" ${MOTION_IS_TOP_LEVEL})
option(MOTION_WITH_SERIAL "Build serial solver, model/program import and simulator" ON)
option(MOTION_WITH_KDL_REFERENCE "Build independent, test-only KDL FK/J comparison" ON)
option(MOTION_WITH_RUCKIG "Build optional local Ruckig adapter" OFF)
option(MOTION_SANITIZE "Enable address/undefined sanitizers" OFF)
option(MOTION_BUILD_FUZZER "Build Clang libFuzzer program parser (requires sanitizers)" OFF)
if(NOT DEFINED MOTION_COMPONENTS)
  set(MOTION_COMPONENTS "" CACHE STRING "Explicit component roots; empty preserves MOTION_WITH_SERIAL/RUCKIG selection")
endif()
include(cmake/MotionTargets.cmake)
include(cmake/MotionDependencies.cmake)
include(cmake/MotionComponents.cmake)
if(MOTION_COMPONENTS)
  set(MOTION_COMPONENT_ROOTS ${MOTION_COMPONENTS})
else()
  set(MOTION_COMPONENT_ROOTS core contracts math kinematics planning execution tracking platform)
  if(MOTION_WITH_SERIAL)
    list(APPEND MOTION_COMPONENT_ROOTS serial)
  endif()
  if(MOTION_WITH_RUCKIG)
    list(APPEND MOTION_COMPONENT_ROOTS ruckig)
  endif()
endif()
motion_resolve_components(MOTION_SELECTED_COMPONENTS ${MOTION_COMPONENT_ROOTS})
message(STATUS "Motion components: ${MOTION_SELECTED_COMPONENTS}")

if(MOTION_BUILD_FUZZER)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR NOT MOTION_SANITIZE OR NOT "program" IN_LIST MOTION_SELECTED_COMPONENTS OR NOT MOTION_BUILD_TESTS)
    message(FATAL_ERROR "MOTION_BUILD_FUZZER requires Clang, sanitizers, tests and the program component")
  endif()
  add_compile_options(-fsanitize=fuzzer-no-link)
  add_link_options(-fsanitize=fuzzer-no-link)
endif()
if(MOTION_SANITIZE AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

foreach(component core contracts math kinematics planning execution tracking platform)
  if(component IN_LIST MOTION_SELECTED_COMPONENTS)
    motion_header_component(${component})
  endif()
endforeach()
if("ruckig" IN_LIST MOTION_SELECTED_COMPONENTS)
  motion_ruckig_dependency()
  motion_header_component(ruckig)
  target_link_libraries(robot_motion_ruckig INTERFACE ruckig)
  target_compile_definitions(robot_motion_ruckig INTERFACE MOTION_WITH_RUCKIG=1)
endif()
add_subdirectory(src)
if("following" IN_LIST MOTION_SELECTED_COMPONENTS)
  motion_header_component(following)
endif()
add_library(robot_motion INTERFACE)
add_library(RobotMotion::motion ALIAS robot_motion)
foreach(component IN LISTS MOTION_COMPONENT_ROOTS)
  target_link_libraries(robot_motion INTERFACE RobotMotion::${component})
endforeach()
if(MOTION_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
if(MOTION_BUILD_EXAMPLES)
  add_subdirectory(examples)
endif()
