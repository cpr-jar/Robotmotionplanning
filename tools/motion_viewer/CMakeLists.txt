cmake_minimum_required(VERSION 3.20)
project(robot_motion_viewer VERSION 0.5.0 LANGUAGES CXX)
get_filename_component(ROBOT_MOTION_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
# This standalone tool embeds the library; it does not join its implementation/build graph.
set(MOTION_BUILD_TESTS OFF)
set(MOTION_BUILD_EXAMPLES OFF)
set(MOTION_WITH_SERIAL ON)
set(MOTION_WITH_RUCKIG OFF)
set(MOTION_WITH_KDL_REFERENCE OFF)
set(MOTION_COMPONENTS program)
add_subdirectory("${ROBOT_MOTION_SOURCE_DIR}" motion_library)
add_executable(motion_viewer_exporter native/export_program_scene.cpp)
target_link_libraries(motion_viewer_exporter PRIVATE RobotMotion::program)
if(NOT MSVC)
  target_compile_options(motion_viewer_exporter PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
option(MOTION_VIEWER_BUILD_TESTS "Enable standalone viewer tool checks" ON)
if(MOTION_VIEWER_BUILD_TESTS)
  enable_testing()
  find_package(Python3 3.9 COMPONENTS Interpreter REQUIRED)
  add_test(NAME motion_viewer_tool_tests COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -v)
  set_tests_properties(motion_viewer_tool_tests PROPERTIES TIMEOUT 600 ENVIRONMENT "MOTION_VIEWER_EXPORTER=$<TARGET_FILE:motion_viewer_exporter>;PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR}")
endif()
