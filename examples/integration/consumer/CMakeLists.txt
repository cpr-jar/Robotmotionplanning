cmake_minimum_required(VERSION 3.20)
project(controller_consumer LANGUAGES CXX)
if(NOT ROBOT_MOTION_SOURCE_DIR)
  message(FATAL_ERROR "Set ROBOT_MOTION_SOURCE_DIR to the Robot Motion source directory")
endif()
add_subdirectory(${ROBOT_MOTION_SOURCE_DIR} robot_motion)
if(TARGET core_tests OR TARGET program_simulator OR TARGET motion_kdl_reference)
  message(FATAL_ERROR "Embedding must not build motion tests, examples or KDL by default")
endif()
add_executable(controller_consumer main.cpp)
target_link_libraries(controller_consumer PRIVATE RobotMotion::motion)
