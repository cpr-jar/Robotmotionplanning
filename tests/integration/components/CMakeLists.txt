cmake_minimum_required(VERSION 3.20)
project(component_consumer LANGUAGES CXX)
if(NOT ROBOT_MOTION_SOURCE_DIR OR NOT CASE)
  message(FATAL_ERROR "Set ROBOT_MOTION_SOURCE_DIR and CASE")
endif()
set(MOTION_BUILD_TESTS OFF)
set(MOTION_BUILD_EXAMPLES OFF)
set(MOTION_COMPONENTS ${CASE})
set(component ${CASE})
if(CASE STREQUAL "serial_planning")
  set(MOTION_COMPONENTS planning serial_kinematics)
  set(component planning)
endif()
add_subdirectory(${ROBOT_MOTION_SOURCE_DIR} motion)
foreach(forbidden core_tests program_simulator motion_kdl_reference)
  if(TARGET ${forbidden})
    message(FATAL_ERROR "Unexpected test/example dependency: ${forbidden}")
  endif()
endforeach()
# This fixture is outside the library directory: inspect exported targets,
# not variables scoped inside that directory.
set(known core contracts math kinematics planning execution tracking platform
  serial_kinematics differential_kinematics calibration dxf_import model_import program ruckig following serial)
set(present "")
foreach(name IN LISTS known)
  if(TARGET RobotMotion::${name})
    list(APPEND present ${name})
  endif()
endforeach()
file(WRITE ${CMAKE_BINARY_DIR}/components.txt "${present}\n")
if(NOT CASE MATCHES "^(model_import|program|serial)$")
  foreach(forbidden yaml-cpp tinyxml2 robot_motion_model_import robot_motion_program robot_motion_serial)
    if(TARGET ${forbidden})
      message(FATAL_ERROR "Unrequested parser/bundle dependency: ${forbidden}")
    endif()
  endforeach()
endif()
if(CASE MATCHES "^(execution|planning|dxf_import)$")
  foreach(forbidden motion_eigen_headers ruckig robot_motion_serial_kinematics robot_motion_calibration)
    if(TARGET ${forbidden})
      message(FATAL_ERROR "Unrequested numerical dependency: ${forbidden}")
    endif()
  endforeach()
endif()
set(source ${CASE}.cpp)
if(CASE STREQUAL "serial")
  set(source program.cpp)
endif()
add_executable(component_consumer ${source})
target_link_libraries(component_consumer PRIVATE RobotMotion::${component})
if(CASE STREQUAL "serial_planning")
  target_link_libraries(component_consumer PRIVATE RobotMotion::serial_kinematics)
endif()
target_compile_definitions(component_consumer PRIVATE MOTION_SOURCE_DIR="${ROBOT_MOTION_SOURCE_DIR}")
if(MSVC)
  target_compile_options(component_consumer PRIVATE /W4 /WX /UNDEBUG)
else()
  target_compile_options(component_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
endif()
